```python
#!/usr/bin/env python3
"""
ez_mv_core.py — Core beat-sync MV composition logic.
No globals, no I/O side effects. All config passed as arguments.
"""

import json
import random
import subprocess
import tempfile
from pathlib import Path

import librosa

TARGET_FPS = 30
OUTPUT_SIZE = "1920:1080"
OUTPUT_SIZE_LAVFI = "1920x1080"
FLASH_DURATION = 0.04


def parse_timecode(tc: str) -> float:
    """'0:15' or '1:05:30' → seconds as float"""
    parts = tc.split(":")
    if len(parts) == 2:
        return int(parts[0]) * 60 + float(parts[1])
    return int(parts[0]) * 3600 + int(parts[1]) * 60 + float(parts[2])


def run(cmd: list):
    subprocess.run(cmd, check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)


def probe_duration(path: Path) -> float:
    result = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration",
         "-of", "default=noprint_wrappers=1:nokey=1", str(path)],
        capture_output=True, text=True, check=True
    )
    return float(result.stdout.strip())


def extract_thumbnail(source: Path, output: Path, at: float = 1.0):
    """Extract a single JPEG frame for UI preview."""
    run([
        "ffmpeg", "-y", "-ss", str(at), "-i", str(source),
        "-vframes", "1", "-q:v", "5",
        "-vf", "scale=320:180:force_original_aspect_ratio=decrease,pad=320:180:(ow-iw)/2:(oh-ih)/2",
        str(output)
    ])


def detect_beats(music_path: Path) -> tuple[list[float], float]:
    """Returns (beat_times, tempo_bpm)."""
    y, sr = librosa.load(str(music_path))
    tempo, beat_frames = librosa.beat.beat_track(y=y, sr=sr)
    beat_times = librosa.frames_to_time(beat_frames, sr=sr).tolist()
    tempo_val = float(tempo) if tempo.ndim == 0 else float(tempo[0])
    return beat_times, tempo_val


def apply_cut_density(beat_times: list[float], density: float) -> list[float]:
    """
    density: float where 1.0 = every beat, 0.25 = every 4th beat, 2.0 = every half-beat.
    Values < 1.0 subsample. Values > 1.0 interpolate midpoints.
    """
    if density <= 0:
        return beat_times
    if density <= 1.0:
        step = max(1, round(1.0 / density))
        return beat_times[::step]
    else:
        # Interpolate: insert n midpoints between each pair of beats
        n_interp = round(density) - 1
        result = []
        for i in range(len(beat_times) - 1):
            result.append(beat_times[i])
            for j in range(1, n_interp + 1):
                t = beat_times[i] + (beat_times[i + 1] - beat_times[i]) * j / (n_interp + 1)
                result.append(t)
        result.append(beat_times[-1])
        return result


def load_philm_offset(analysis_dir: Path, clip_stem: str) -> float | None:
    """
    Check for philm affect/sizzle JSON at analysis_dir/clip_stem.json.
    Returns the timecode (seconds) of the highest-intensity moment, or None.
    """
    json_path = analysis_dir / f"{clip_stem}.json"
    if not json_path.exists():
        return None
    try:
        data = json.loads(json_path.read_text())
        timecodes = data.get("timecodes", [])
        if not timecodes:
            return None
        # Works for affect (intensity) and sizzle (energy) schemas
        best = max(timecodes, key=lambda m: m.get("intensity", m.get("energy", 0)))
        return parse_timecode(best["time"])
    except Exception:
        return None


def build_cut_schedule(
    beat_times: list[float],
    clips: list[dict],          # [{"path": Path, "offset": float}, ...]
    video_duration: float,
    flash_prob: float,
    stutter_prob: float,
    seed: int | None = None,
) -> tuple[list[dict], int]:
    """
    Returns (schedule, seed_used).
    Each schedule entry: {path, start_offset, duration, flash_before, stutter, beat_time}
    """
    if seed is None:
        seed = random.randint(0, 9999)
    rng = random.Random(seed)

    pool = list(clips)
    rng.shuffle(pool)

    # Filter beats to track duration
    beat_times = [t for t in beat_times if t < video_duration]

    schedule = []
    pool_idx = 0

    for i, beat_t in enumerate(beat_times):
        duration = beat_times[i + 1] - beat_t if i + 1 < len(beat_times) else 0.5
        if duration < FLASH_DURATION:
            continue

        stutter = rng.random() < stutter_prob
        if not stutter or len(schedule) == 0:
            clip = pool[pool_idx % len(pool)]
            pool_idx += 1
        else:
            prev = schedule[-1]
            clip = {"path": prev["path"], "offset": prev["start_offset"] + prev["duration"]}

        schedule.append({
            "path": clip["path"],
            "start_offset": clip["offset"],
            "duration": duration,
            "flash_before": rng.random() < flash_prob,
            "stutter": stutter,
            "beat_time": beat_t,
        })

    return schedule, seed


def generate_flash_frame(tmp_dir: Path) -> Path:
    white = tmp_dir / "white_flash.mp4"
    run([
        "ffmpeg", "-y", "-f", "lavfi",
        "-i", f"color=white:s={OUTPUT_SIZE_LAVFI}:d={FLASH_DURATION:.4f}",
        "-c:v", "libx264", "-preset", "fast", "-crf", "18",
        "-r", str(TARGET_FPS), str(white)
    ])
    return white


def extract_segment(source: Path, start: float, duration: float, output: Path):
    run([
        "ffmpeg", "-y",
        "-ss", f"{start:.4f}", "-t", f"{duration:.4f}",
        "-i", str(source),
        "-vf", (
            f"scale={OUTPUT_SIZE}:force_original_aspect_ratio=decrease,"
            f"pad={OUTPUT_SIZE}:(ow-iw)/2:(oh-ih)/2,"
            f"fps={TARGET_FPS}"
        ),
        "-c:v", "libx264", "-preset", "fast", "-crf", "18",
        "-an", str(output)
    ])


def assemble(
    schedule: list[dict],
    white_flash: Path,
    music_path: Path,
    output_path: Path,
    video_duration: float,
    progress_callback=None,
):
    """
    Full assembly pipeline. progress_callback(current, total) called per segment.
    """
    with tempfile.TemporaryDirectory(prefix="ez_mv_") as tmp:
        tmp_dir = Path(tmp)
        segs_dir = tmp_dir / "segs"
        segs_dir.mkdir()

        ordered_segments: list[Path] = []
        playhead: dict[Path, float] = {}

        # Initialize playheads from first occurrence of each clip
        seen = set()
        for entry in schedule:
            p = entry["path"]
            if p not in seen:
                playhead[p] = entry["start_offset"]
                seen.add(p)

        total = len(schedule)
        for idx, entry in enumerate(schedule):
            if entry["flash_before"]:
                ordered_segments.append(white_flash)

            seg_path = segs_dir / f"seg_{idx:05d}.mp4"
            src = entry["path"]
            start = playhead[src]
            duration = entry["duration"]

            src_dur = probe_duration(src)
            if start + duration > src_dur:
                start = max(0.0, src_dur - duration - 0.1)

            extract_segment(src, start, duration, seg_path)
            ordered_segments.append(seg_path)

            if not entry["stutter"]:
                playhead[src] = start + duration

            if progress_callback:
                progress_callback(idx + 1, total)

        # Write concat list
        concat_file = tmp_dir / "concat.txt"
        with open(concat_file, "w") as f:
            for seg in ordered_segments:
                f.write(f"file '{seg.absolute()}'\n")

        # Final assemble with music
        subprocess.run([
            "ffmpeg", "-y",
            "-f", "concat", "-safe", "0", "-i", str(concat_file),
            "-i", str(music_path),
            "-map", "0:v", "-map", "1:a",
            "-c:v", "libx264", "-preset", "slow", "-crf", "18",
            "-c:a", "aac", "-b:a", "192k",
            "-t", str(video_duration),
            "-movflags", "+faststart",
            str(output_path)
        ], check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
```

```python
#!/usr/bin/env python3
"""
EZ_MV — Beat-sync music video composer.
Run: uv run streamlit run scripts/ez_mv_app.py
"""

import json
import tempfile
from pathlib import Path

import streamlit as st

from ez_mv_core import (
    apply_cut_density,
    assemble,
    detect_beats,
    extract_thumbnail,
    generate_flash_frame,
    build_cut_schedule,
    load_philm_offset,
    probe_duration,
)

st.set_page_config(page_title="EZ_MV", page_icon="🎬", layout="wide")
st.title("EZ_MV")
st.caption("Beat-sync music video composer")

# ─── SIDEBAR: SETTINGS ─────────────────────────────────────────────────────────

with st.sidebar:
    st.header("Settings")
    output_dir = Path(st.text_input("Output folder", value=str(Path.home() / "ez_mv_output")))
    output_dir.mkdir(parents=True, exist_ok=True)

    philm_dir_input = st.text_input("Philm analysis folder (optional)", value="")
    philm_dir = Path(philm_dir_input) if philm_dir_input.strip() else None

# ─── MUSIC TRACK ───────────────────────────────────────────────────────────────

st.header("1. Music Track")
music_folder = st.text_input("Music folder", value=str(Path.home()))
music_path = None
music_files = []

if music_folder and Path(music_folder).is_dir():
    music_files = sorted(
        p for p in Path(music_folder).iterdir()
        if p.suffix.lower() in (".wav", ".mp3", ".aiff", ".m4a", ".flac")
    )

if music_files:
    selected_music = st.selectbox(
        "Select track",
        options=music_files,
        format_func=lambda p: p.name,
    )
    music_path = selected_music
    st.caption(f"`{music_path}`")
else:
    st.info("Enter a folder path above containing audio files.")

# ─── CLIP POOL ─────────────────────────────────────────────────────────────────

st.header("2. Clips")
clips_folder = st.text_input("Clips folder")
VIDEO_EXTS = {".mov", ".mp4", ".mxf", ".avi", ".mkv", ".m4v"}

clips_config: list[dict] = []  # [{path, offset, enabled}]

if clips_folder and Path(clips_folder).is_dir():
    clip_files = sorted(
        p for p in Path(clips_folder).iterdir()
        if p.suffix.lower() in VIDEO_EXTS
    )

    if not clip_files:
        st.warning("No video files found in that folder.")
    else:
        st.caption(f"{len(clip_files)} clips found")

        # Generate thumbnails (cached by session state)
        if "thumbnails" not in st.session_state or st.session_state.get("thumb_dir_key") != clips_folder:
            st.session_state["thumb_dir_key"] = clips_folder
            thumb_dir = Path(tempfile.mkdtemp(prefix="ez_mv_thumbs_"))
            thumbs = {}
            with st.spinner("Generating previews..."):
                for clip in clip_files:
                    thumb_path = thumb_dir / f"{clip.stem}.jpg"
                    try:
                        extract_thumbnail(clip, thumb_path)
                        thumbs[clip] = thumb_path
                    except Exception:
                        thumbs[clip] = None
            st.session_state["thumbnails"] = thumbs

        thumbs = st.session_state["thumbnails"]

        # Render clip checklist in 4-column grid
        cols_per_row = 4
        rows = [clip_files[i:i+cols_per_row] for i in range(0, len(clip_files), cols_per_row)]

        for row in rows:
            cols = st.columns(cols_per_row)
            for col, clip in zip(cols, row):
                with col:
                    # Thumbnail
                    if thumbs.get(clip):
                        st.image(str(thumbs[clip]), use_container_width=True)
                    else:
                        st.markdown("*(no preview)*")

                    # Checkbox
                    enabled = st.checkbox(clip.stem, value=True, key=f"chk_{clip}")

                    # Auto-fill offset from philm if available
                    auto_offset = None
                    if philm_dir and philm_dir.is_dir():
                        auto_offset = load_philm_offset(philm_dir, clip.stem)

                    default_offset = auto_offset if auto_offset is not None else 0.0
                    label = f"offset (s)" + (" ✦" if auto_offset is not None else "")
                    offset = st.number_input(
                        label, min_value=0.0, value=default_offset,
                        step=0.5, key=f"off_{clip}"
                    )

                    if enabled:
                        clips_config.append({"path": clip, "offset": offset})

# ─── CUT PARAMETERS ────────────────────────────────────────────────────────────

st.header("3. Cut Parameters")

col1, col2, col3 = st.columns(3)

with col1:
    cut_density = st.slider(
        "Cut density",
        min_value=0.25, max_value=2.0, value=1.0, step=0.25,
        help="0.25 = every 4 beats (slow) · 1.0 = every beat · 2.0 = every half-beat (chaos)"
    )
    density_label = {0.25: "every 4 beats", 0.5: "every 2 beats", 1.0: "every beat", 2.0: "every half-beat"}
    st.caption(density_label.get(cut_density, f"{cut_density}x"))

with col2:
    flash_prob = st.slider(
        "Flash frequency",
        min_value=0.0, max_value=0.5, value=0.25, step=0.05,
        format="%.0f%%",
        help="Probability of white flash frame before each cut"
    )
    st.caption(f"{flash_prob*100:.0f}% of cuts")

with col3:
    stutter_prob = st.slider(
        "Stutter frequency",
        min_value=0.0, max_value=0.3, value=0.15, step=0.05,
        format="%.0f%%",
        help="Probability of repeating a clip back-to-back"
    )
    st.caption(f"{stutter_prob*100:.0f}% of cuts")

# ─── GENERATE ──────────────────────────────────────────────────────────────────

st.header("4. Generate")

ready = music_path and len(clips_config) >= 2
if not ready:
    st.info("Select a music track and at least 2 clips to continue.")

if ready and st.button("⚡ Generate MV", type="primary", use_container_width=True):
    video_duration = probe_duration(music_path)

    with st.status("Generating...", expanded=True) as status:
        st.write(f"Music: `{music_path.name}` ({video_duration:.1f}s)")

        # Beat detection
        st.write("Detecting beats...")
        raw_beats, tempo = detect_beats(music_path)
        beat_times = apply_cut_density(raw_beats, cut_density)
        st.write(f"✓ {tempo:.1f} BPM → {len(beat_times)} cut points at {cut_density}x density")

        # Cut schedule
        schedule, seed = build_cut_schedule(
            beat_times, clips_config, video_duration, flash_prob, stutter_prob
        )
        flash_n = sum(1 for s in schedule if s["flash_before"])
        stutter_n = sum(1 for s in schedule if s["stutter"])
        st.write(f"✓ {len(schedule)} cuts — {flash_n} flashes, {stutter_n} stutters (seed: {seed})")

        # Versioned output path
        existing = sorted(output_dir.glob("ez_mv_v*.mp4"))
        next_v = len(existing) + 1
        output_path = output_dir / f"ez_mv_v{next_v:02d}.mp4"
        st.write(f"Output: `{output_path}`")

        # Flash frame
        tmp_flash_dir = Path(tempfile.mkdtemp(prefix="ez_mv_flash_"))
        white_flash = generate_flash_frame(tmp_flash_dir)

        # Segment extraction + assembly
        progress_bar = st.progress(0)
        progress_text = st.empty()

        def on_progress(current, total):
            progress_bar.progress(current / total)
            progress_text.text(f"Extracting segment {current}/{total}")

        assemble(schedule, white_flash, music_path, output_path, video_duration, on_progress)

        progress_bar.progress(1.0)
        progress_text.text("Done!")
        status.update(label="✓ Complete!", state="complete")

    st.success(f"Saved to `{output_path}`")

    # Save session metadata
    meta = {
        "output": str(output_path),
        "music": str(music_path),
        "seed": seed,
        "tempo": tempo,
        "cuts": len(schedule),
        "cut_density": cut_density,
        "flash_prob": flash_prob,
        "stutter_prob": stutter_prob,
        "clips": [str(c["path"]) for c in clips_config],
    }
    meta_path = output_path.with_suffix(".json")
    meta_path.write_text(json.dumps(meta, indent=2))
    st.caption(f"Session saved to `{meta_path}`")
```
